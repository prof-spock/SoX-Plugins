96 String st = "[";
97 const Natural effectiveCount =

99
100 for (Natural i = 0; i < effectiveCount; i++) {

104 " : "

108 st += (_length > dataPointCount ? ", ..." : "");
109 st += "]";

115 result += ", _data = " + st;
