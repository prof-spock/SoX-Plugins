220/*--------------------*/
221/* setup */
222/*--------------------*/
223
224INLINE
225void _WaveFormDescriptor::set (Real length, WaveFormKind kind,
226 Real minimumValue, Real maximumValue,
227 Radians phase, Boolean hasIntegerValues)
228{
229 Logging_trace6(">>: length = %1, kind = %2,"
230 " minimumValue = %3, maximumValue = %4,"
231 " phase = %5, hasIntegerValues = %6",
232 TOSTRING(length), waveFormKindToString(kind),
233 TOSTRING(minimumValue), TOSTRING(maximumValue),
234 TOSTRING(phase), TOSTRING(hasIntegerValues));
235 Assertion_pre(length > 0.0, _ErrMsg_badLength);
236
237 _WaveTablePtr buffer = (kind == WaveFormKind::sine
238 ? &_sineWaveTable
239 : &_triangleWaveTable);
240 const Natural interpolationOrder =
241 (kind == WaveFormKind::sine ? 3 : 1);
242 const Natural waveTableLength = buffer->length();
243 Assertion_pre(waveTableLength > 0, _ErrMsg_badWaveTable);
244
245 const Real firstPosition =
246 Real::mod((Real) waveTableLength * phase / Real::twoPi,
247 (Real) waveTableLength);
248
249 _kind = kind;
250 _firstPosition = firstPosition;
251 _increment = (Real) waveTableLength / length;
252 _stepCount = 0;
253 _position = firstPosition;
254 _minimumValue = minimumValue;
255 _maximumValue = maximumValue;
256 _hasIntegerValues = hasIntegerValues;
257 _waveTableLength = waveTableLength;
258 _buffer = buffer;
259
260 _interpolator.set(*buffer, waveTableLength, Real{waveTableLength},
261 true, interpolationOrder);
262
263 Logging_trace("<<");
264}

266/*--------------------*/
267/* property access */
268/*--------------------*/
269
270INLINE
271Real _WaveFormDescriptor::current () const
272{
273 const Real scalingFactor = _maximumValue - _minimumValue;
274 Assertion_check(_position >= 0.0, "position must be non-negative");
275 Real value = _interpolator.at(_position);
276 value = _minimumValue + value * scalingFactor;
277
278 /* the following does not distribute integer values equally over
279 their range: the minimum and maximum value only get 50%
280 coverage of all the other values */
281 value = (_hasIntegerValues ? value.round() : value);
282 return value;
283}

524 TOSTRING(frequency), TOSTRING(timeOffset),
525 TOSTRING(currentTime));
526
527 Radians result;
